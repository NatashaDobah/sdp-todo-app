Yes, more reusable and type-safe than my snippet — full CRUD surface, no more formData.get() casting, and it matches your Task type properly. But there's one bug that will break your build:

export { isOverdue } from './tasks'; will fail. Once a file has 'use server' at the top, Next.js requires every export to be an async function (since every export is treated as a callable server action). isOverdue is a synchronous pure function, not a server action — it doesn't touch the database — so this line will throw a build error like "only async functions are allowed to be exported in a 'use server' file."

The fix is really an improvement anyway: isOverdue has no business being in a server-only file at all. It doesn't read fs, path, or the DB — it's just date logic. Pull it out into something like lib/utils.ts (no 'use server' directive) and import it straight into your client components. That way you're not even paying a server round-trip to check if something's overdue on every render.

One other thing worth thinking about, not a bug but a design tradeoff: getActiveTasksAction, getArchivedTasksAction, getTasksSortedByAction being server actions means if page.tsx stays a client component, you'll fetch data via useEffect + these actions — which works, but means no server-rendered initial data (a waterfall: page loads empty, then fetches). The alternative is making the list-rendering part a server component that calls dbGetActiveTasks() directly (no action wrapper needed for reads in a server component), and only pulling the form/buttons into a small client component that calls your mutation actions (createTaskAction, updateTaskAction, archiveTaskAction). That's a bit more restructuring but is the more idiomatic App Router pattern and avoids the extra round trip.

Given time constraints, do you want to keep it simple (client component + these fetch actions, accept the waterfall) or restructure into server/client split? Either is defensible — just make sure whichever one you pick, you can explain why in your docs, since "why this architecture" is exactly the kind of thing that shows up in viva/marking discussions.

import { getActiveTasks, getArchivedTasks, getTasksSortedBy } from '../lib/tasks';
import { TaskForm } from './TaskForm';
import { TaskControls } from './TaskControls';
import { TaskItem } from './TaskItem';

export default async function Home({
  searchParams,
}: {
  searchParams: Promise<{ view?: string; sort?: string }>;
}) {
  const { view, sort } = await searchParams;
  const showArchived = view === 'archived';
  const sortBy = (sort as 'topic' | 'status' | 'due_date') || 'due_date';

  const tasks = showArchived
    ? await getArchivedTasks()
    : await getTasksSortedBy(sortBy);

  return (
    <main>
      <TaskForm />
      <TaskControls showArchived={showArchived} sortBy={sortBy} />
      <ul>
        {tasks.map((task) => (
          <TaskItem key={task.id} task={task} showArchived={showArchived} />
        ))}
      </ul>
    </main>
  );

'use client';
import { updateTaskAction, archiveTaskAction } from '../../lib/actions';
import { isOverdue } from '../../lib/utils';
import { Task } from '../../lib/tasks';

export function TaskItem({ task, showArchived }: { task: Task; showArchived: boolean }) {
  async function handleStatusChange() {
    const nextStatus = task.status === 'Todo' ? 'In-Progress' :
                       task.status === 'In-Progress' ? 'Complete' : 'Todo';
    await updateTaskAction(task.id, { status: nextStatus as 'Todo' | 'In-Progress' | 'Complete' });
  }

  async function handleArchive() {
    await archiveTaskAction(task.id);
  }

  return (
    <li>
      <article className={`bg-white rounded-lg shadow p-4 border-l-4 ${
        isOverdue(task) && task.status !== 'Complete'
          ? 'border-red-500 bg-red-50'
          : 'border-blue-500'
      }`}>
        <header className="flex justify-between items-start">
          <hgroup className="flex-1">
            <h3 className="font-semibold text-lg">{task.title}</h3>
            {task.description && <p className="text-gray-600">{task.description}</p>}
          </hgroup>
          <menu className="flex space-x-2">
            {!showArchived && (
              <>
                <button
                  onClick={handleStatusChange}
                  className="bg-green-500 hover:bg-green-700 text-white text-sm px-3 py-1 rounded"
                  aria-label={`Change status of "${task.title}"`}
                >
                  {task.status === 'Todo' ? 'Start' :
                   task.status === 'In-Progress' ? 'Complete' : 'Reset'}
                </button>
                <button
                  onClick={handleArchive}
                  className="bg-yellow-500 hover:bg-yellow-700 text-white text-sm px-3 py-1 rounded"
                  aria-label={`Archive "${task.title}"`}
                >
                  Archive
                </button>
              </>
            )}
          </menu>
        </header>
        <footer className="flex flex-wrap gap-2 mt-1 text-sm text-gray-500">
          <time className="text-gray-600" dateTime={task.due_date}>📅 {task.due_date}</time>
          <span className="sr-only">Topic:</span>
          <span className="text-gray-600">📂 {task.topic}</span>
          <span className="sr-only">Status:</span>
          <span className="text-gray-600">📌 {task.status}</span>
          {isOverdue(task) && task.status !== 'Complete' && (
            <strong className="text-red-500 font-bold">⚠️ OVERDUE</strong>
          )}
        </footer>
      </article>
    </li>
  );
}
